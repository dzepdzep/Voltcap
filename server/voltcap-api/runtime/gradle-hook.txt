
// (Batch 518, DEV ONLY — never committed) Runs server/voltcap-api/runtime/task.ps1 after :app:compileDebugKotlin while
// server/voltcap-api/runtime/.run-now exists (its first line names the action). Fake credentials only.
if (rootProject.file("server/voltcap-api/runtime/.run-now").isFile) {

        commandLine("powershell.exe", "-NoProfile", "-ExecutionPolicy", "Bypass", "-File",

        isIgnoreExitValue = true
    }
    tasks.matching { it.name == "compileDebugKotlin" }.configureEach { finalizedBy(workerTask) }
